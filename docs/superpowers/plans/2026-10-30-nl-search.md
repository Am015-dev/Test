# NL Search Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A dashboard page where the user types what home they want in free Greek text and sees existing listings ranked by match %.

**Architecture:** One new module `scout/nl_search.py` (pydantic `SearchFilters`, pure `match()`, `parse_query()` calling Claude with a forced tool call). One new page function in `ui/app.py` registered in `PAGES`. No DB/schema/collector/scoring changes.

**Tech Stack:** Python 3.11, pydantic v2, `anthropic` SDK, Streamlit, pytest.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-10-30-nl-search-design.md` (source of truth).
- `from __future__ import annotations`; line length 100 (black + ruff).
- Query text is cut to 500 chars; max 20 API calls per Streamlit session.
- API failure / invalid output must raise `SearchParseError`, never fall back to "all listings".
- Missing `ANTHROPIC_API_KEY` → page shows an instruction, makes no call, nothing crashes.
- Unknown listing fields never exclude a listing. Listing dicts (from `queries.load_listings`)
  have keys: `price, area_m2, bedrooms, floor, renovated_year, neighborhood, address,
  search_area, title, is_private_seller, url, source, price_per_m2`. No description field.
- Model id: `claude-haiku-5-5`.
- Verification gate before the final commit: `python -m pytest -q`, `ruff check src/ tests/ scripts/`,
  `black --check src/ tests/ scripts/`.

```python
"""Free-text home search: text -> SearchFilters (Claude) -> match % over listing dicts."""

from __future__ import annotations

import unicodedata
from typing import Literal

from pydantic import BaseModel, Field

Want = Literal["elevator", "parking", "balcony", "renovated"]

# The listing dict has no feature columns, so wants are matched on title/location text.
WANT_WORDS: dict[str, tuple[str, ...]] = {
    "elevator": ("ανελκυστηρ", "ασανσερ", "elevator", "lift"),
    "parking": ("παρκινγκ", "γκαραζ", "σταθμευσ", "parking", "garage"),
    "balcony": ("μπαλκον", "βεραντ", "balcon", "terrace"),
    "renovated": ("ανακαινισ", "renovat"),
}
WANT_LABEL = {
    "elevator": "ανελκυστήρας",
    "parking": "πάρκινγκ",
    "balcony": "μπαλκόνι",
    "renovated": "ανακαινισμένο",
}


class SearchFilters(BaseModel):
    areas: list[str] = Field(
        default_factory=list,
        description="Neighbourhoods/municipalities. Give BOTH the Greek name and its Latin "
        "transliteration as separate entries (e.g. 'Χολαργός', 'Cholargos').",
    )
    price_min: int | None = None
    price_max: int | None = None
    area_min: float | None = Field(None, description="Min square metres")
    area_max: float | None = None
    bedrooms_min: int | None = Field(None, description="'δυάρι' = 1 bedroom, 'τριάρι' = 2")
    bedrooms_max: int | None = None
    floor_min: float | None = None
    private_only: bool = False
    wants: list[Want] = Field(default_factory=list)


def _norm(s: str) -> str:
    s = unicodedata.normalize("NFD", s.casefold())
    return "".join(c for c in s if not unicodedata.combining(c))


def match(d: dict, f: SearchFilters) -> tuple[int, list[str]] | None:
    """Return (match %, reasons), or None if a hard criterion fails.

    Unknown (None) listing fields never exclude. Wants are soft: 60 + 40 * confirmed/total.
    """
    reasons: list[str] = []
    where = _norm(" ".join(str(d.get(k) or "") for k in ("neighborhood", "address", "search_area")))
    if f.areas:
        hit = next((a for a in f.areas if _norm(a) in where), None)
        if hit is None:
            return None
        reasons.append(f"περιοχή: {hit}")
    bounds = (
        ("price", f.price_min, f.price_max, "τιμή"),
        ("area_m2", f.area_min, f.area_max, "τ.μ."),
        ("bedrooms", f.bedrooms_min, f.bedrooms_max, "υπνοδωμάτια"),
        ("floor", f.floor_min, None, "όροφος"),
    )
    for key, lo, hi, label in bounds:
        v = d.get(key)
        if v is None:
            continue
        if (lo is not None and v < lo) or (hi is not None and v > hi):
            return None
        if lo is not None or hi is not None:
            reasons.append(f"{label}: {v:g}")
    if f.private_only and d.get("is_private_seller") is False:
        return None
    text = _norm(f"{where} {d.get('title') or ''}")
    confirmed = [
        w
        for w in f.wants
        if any(k in text for k in WANT_WORDS[w]) or (w == "renovated" and d.get("renovated_year"))
    ]
    reasons += [WANT_LABEL[w] for w in confirmed]
    pct = 100 if not f.wants else 60 + 40 * len(confirmed) // len(f.wants)
    return pct, reasons
```

from types import SimpleNamespace

import pytest

from scout.nl_search import MAX_CHARS, SearchParseError, parse_query


def _client(content=None, boom=None):
    calls = []

    def create(**kw):
        calls.append(kw)
        if boom:
            raise boom
        return SimpleNamespace(content=content)

    return SimpleNamespace(messages=SimpleNamespace(create=create)), calls


def test_parse_query_returns_filters_and_forces_tool():
    block = SimpleNamespace(type="tool_use", input={"areas": ["Χολαργός"], "price_max": 180000})
    client, calls = _client([block])
    f = parse_query("δυάρι στον Χολαργό έως 180Χ", client=client)
    assert f.areas == ["Χολαργός"] and f.price_max == 180_000
    assert calls[0]["tool_choice"] == {"type": "tool", "name": "set_filters"}


def test_parse_query_truncates_text():
    client, calls = _client([SimpleNamespace(type="tool_use", input={})])
    parse_query("α" * 900, client=client)
    assert len(calls[0]["messages"][0]["content"]) == MAX_CHARS


@pytest.mark.parametrize(
    "content,boom",
    [
        ([SimpleNamespace(type="text", text="no tool")], None),
        ([SimpleNamespace(type="tool_use", input={"price_max": "lots"})], None),
        (None, RuntimeError("api down")),
    ],
)
def test_parse_query_failures_raise(content, boom):
    client, _ = _client(content, boom)
    with pytest.raises(SearchParseError):
        parse_query("κάτι", client=client)


def test_parse_query_empty_or_missing_key(monkeypatch):

```python
MAX_CHARS = 500
MODEL = "claude-haiku-5-5"
SYSTEM = (
    "Convert a Greek home-BUYING search request into filters by calling set_filters. "
    "Prices are purchase prices in EUR ('180Χ' = 180000). Only set fields the user stated."
)


class SearchParseError(Exception):
    """The query could not be turned into filters (shown to the user, never swallowed)."""


def parse_query(text: str, client=None) -> SearchFilters:
    text = text.strip()[:MAX_CHARS]
    if not text:
        raise SearchParseError("Γράψε τι σπίτι ψάχνεις.")
    if client is None:
        key = get_settings().anthropic_api_key
        if not key:
            raise SearchParseError("Λείπει το ANTHROPIC_API_KEY (βάλ'το στο .env ή στα secrets).")
        import anthropic

        client = anthropic.Anthropic(api_key=key)
    try:
        resp = client.messages.create(
            model=MODEL,
            max_tokens=500,
            system=SYSTEM,
            messages=[{"role": "user", "content": text}],
            tools=[
                {
                    "name": "set_filters",
                    "description": "Set the search filters.",
                    "input_schema": SearchFilters.model_json_schema(),
                }
            ],
            tool_choice={"type": "tool", "name": "set_filters"},
        )
        block = next(b for b in resp.content if b.type == "tool_use")
        return SearchFilters.model_validate(block.input)
    except Exception as e:  # API, no tool block, or invalid filters: all the same to the user
        raise SearchParseError(f"Δεν κατάλαβα την αναζήτηση ({type(e).__name__}).") from e
```

```python
MAX_SEARCHES = 20  # per Streamlit session — bounds API cost


def page_search(listings: list[dict]) -> None:
    st.subheader("🔎 Γράψε τι σπίτι ψάχνεις")
    text = st.text_area(
        "Με δικά σου λόγια",
        max_chars=nl_search.MAX_CHARS,
        placeholder="δυάρι στον Χολαργό έως 180.000€, από 60 τ.μ., με ανελκυστήρα, μόνο από ιδιώτη",
    )
    used = st.session_state.get("nl_calls", 0)
    if st.button("Αναζήτηση", disabled=not text.strip() or used >= MAX_SEARCHES):
        try:
            st.session_state["nl_filters"] = nl_search.parse_query(text)
            st.session_state["nl_calls"] = used + 1
        except nl_search.SearchParseError as e:
            st.session_state.pop("nl_filters", None)
            st.error(str(e))
    if used >= MAX_SEARCHES:
        st.warning("Έφτασες το όριο αναζητήσεων για αυτή τη συνεδρία — κάνε refresh σελίδας.")
    f = st.session_state.get("nl_filters")
    if f is None:
        return
    st.caption("Κατάλαβα: " + " · ".join(f"{k}={v}" for k, v in f.model_dump(exclude_defaults=True).items()))
    hits = [(m, d) for d in listings if (m := nl_search.match(d, f)) is not None]
    hits.sort(key=lambda h: (-h[0][0], h[1]["price"]))
    st.write(f"**{len(hits)}** αγγελίες ταιριάζουν")
    for (pct, reasons), d in hits[:50]:
        with st.container(border=True):
            st.markdown(
                f"**{pct}%** · [{d.get('title') or d['source_id']}]({d['url']}) — "
                f"{d['price']:,} € · {d.get('area_m2') or '?'} τ.μ. · "
                f"{d.get('price_per_m2') or 0:,.0f} €/τ.μ. · {d['source'].upper()}"
            )
            if reasons:
                st.caption("Γιατί ταιριάζει: " + ", ".join(reasons))
```
